CREATE TABLE Proveedores (
    pr_id   INTEGER      NOT NULL,
    nombre  VARCHAR2(100) NOT NULL,
    ciudad  VARCHAR2(100),

    CONSTRAINT pk_proveedores
        PRIMARY KEY (pr_id)
);


CREATE TABLE Partes (
    pa_id   INTEGER       NOT NULL,
    nombre  VARCHAR2(100) NOT NULL,
    color   VARCHAR2(50),

    CONSTRAINT pk_partes
        PRIMARY KEY (pa_id)
);


CREATE TABLE Catalogo (
    pr_id   INTEGER       NOT NULL,
    pa_id   INTEGER       NOT NULL,
    costo   NUMBER(12,2)  NOT NULL,

    CONSTRAINT pk_catalogo
        PRIMARY KEY (pr_id, pa_id),

    CONSTRAINT fk_catalogo_proveedor
        FOREIGN KEY (pr_id)
        REFERENCES Proveedores (pr_id),

    CONSTRAINT fk_catalogo_parte
        FOREIGN KEY (pa_id)
        REFERENCES Partes (pa_id)
);
